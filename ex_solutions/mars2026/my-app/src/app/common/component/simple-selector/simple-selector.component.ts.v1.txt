import { Component, input, output } from '@angular/core';

@Component({
  selector: 'app-simple-selector',
  imports: [],
  templateUrl: './simple-selector.component.html',
  styleUrl: './simple-selector.component.css',
})
export class SimpleSelectorComponent {
     title=input("titre");
    values=input<string[]>([]);
    choix=output<string>();

    selection="";

    onSelect(val:string){
      this.selection=val;
      this.choix.emit(val);
    }
}
